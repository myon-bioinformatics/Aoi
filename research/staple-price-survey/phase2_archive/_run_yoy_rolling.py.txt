def getf(r, *keys):
    for k in keys:
        if k in r and r[k] not in ("", None):
            return float(r[k])
    raise KeyError(keys)

years = []
for r in rows:
    years.append({
        "year": int(float(r["year"])),
        "P": getf(r, "P_yen", "P"),
        "A": getf(r, "Aoki_A", "Aoki", "A"),
        "E": getf(r, "E"),
        "q": getf(r, "q_kg", "q") if r.get("q_kg") not in (None, "") or r.get("q") not in (None, "") else 5.0,
    })
years.sort(key=lambda d: d["year"])
by_y = {d["year"]: d for d in years}

# Build YoY observations for calendar years with lag
yoy = []  # list of dicts keyed by year t (the end year of the change)
for t in sorted(by_y):
    if t - 1 not in by_y:
        continue
    cur, prev = by_y[t], by_y[t - 1]
    if cur["P"] <= 0 or prev["P"] <= 0 or cur["E"] <= 0 or prev["E"] <= 0:
        continue
    if cur["A"] <= 0 or prev["A"] <= 0:
        continue
    if (100 - cur["A"]) <= 0 or (100 - prev["A"]) <= 0:
        continue
    if (200 - cur["A"]) <= 0 or (200 - prev["A"]) <= 0:
        continue
    rec = {
        "year": t,
        "y": math.log(cur["P"] / prev["P"]),
        "dlogE": math.log(cur["E"] / prev["E"]),
        "dlog100A": math.log((100 - cur["A"]) / (100 - prev["A"])),
        "dlog200A": math.log((200 - cur["A"]) / (200 - prev["A"])),
        "dlogA": math.log(cur["A"] / prev["A"]),
        "neg_dA": -(cur["A"] - prev["A"]),
        "P": cur["P"],
        "A": cur["A"],
        "E": cur["E"],
    }
    yoy.append(rec)

by_yoy = {d["year"]: d for d in yoy}
yoy_years = [d["year"] for d in yoy]  # 2016..2025

MIN_TRAIN = 5
TEST_YEARS = [t for t in range(2021, 2026) if t in by_yoy]

EPS_DEN = 1e-15


def design_row(d, form):
    if form == "aoff_yoy":
        return np.array([d["dlogE"]], dtype=float)
    if form == "primary_yoy":
        return np.array([d["dlog100A"], d["dlogE"]], dtype=float)
    if form == "alt200_yoy":
        return np.array([d["dlog200A"], d["dlogE"]], dtype=float)
    if form == "exp_neg_yoy":
        return np.array([d["neg_dA"], d["dlogE"]], dtype=float)
    if form == "prop_A_yoy":
        return np.array([d["dlogA"], d["dlogE"]], dtype=float)
    raise ValueError(form)


def fit_no_intercept(train, form):
    X = np.vstack([design_row(d, form) for d in train])
    Y = np.array([d["y"] for d in train], dtype=float)
    if form == "aoff_yoy":
        denom = float((X[:, 0] ** 2).sum())
        if abs(denom) < EPS_DEN:
            gamma = 0.0
        else:
            gamma = float((Y * X[:, 0]).sum() / denom)
        return {"alpha": None, "gamma": gamma, "coeffs": np.array([gamma])}
    # 2-regressor through origin
    # normal eq: (X'X) b = X'Y
    XtX = X.T @ X
    XtY = X.T @ Y
    try:
        b, residuals, rank, s = np.linalg.lstsq(X, Y, rcond=None)
        b = np.asarray(b, dtype=float).ravel()
    except Exception:
        b = np.zeros(X.shape[1])
    return {"alpha": float(b[0]), "gamma": float(b[1]), "coeffs": b}


def predict(d, form, fit):
    x = design_row(d, form)
    return float(x @ fit["coeffs"])


win_rows = []
coeff_rows = []

for t in TEST_YEARS:
    train = [d for d in yoy if 2016 <= d["year"] <= t - 1]
    n_train = len(train)
    if n_train < MIN_TRAIN:
        continue  # should not happen for t>=2021
    test_d = by_yoy[t]
    y = test_d["y"]
    for form in FORMS:
        fit = fit_no_intercept(train, form)
        yhat = predict(test_d, form, fit)
        e_m = abs(yhat - y)
        e_n = abs(y)  # zero baseline
        year_win = e_m < e_n
        win_rows.append({
            "t": t,
            "form": form,
            "y": round(y, 8),
            "yhat": round(yhat, 8),
            "abs_e_model": round(e_m, 8),
            "abs_e_naive": round(e_n, 8),
            "year_win": year_win,
            "n_train": n_train,
            "train_end": t - 1,
            "in_excl_aggregate": t != 2025,
            "A": test_d["A"],
            "E": test_d["E"],
            "dlogE": round(test_d["dlogE"], 8),
        })

            "t": t,
            "form": form,
            "train_start": 2016,
            "train_end": t - 1,
            "n_train": n_train,
            "alpha": None if fit["alpha"] is None else round(fit["alpha"], 8),
            "gamma": round(fit["gamma"], 8),
        })

    }
    out_lr.append(rec)
    by_form[r["form"]].append(rec)


def boot_median_ci(xs, B=2000, seed=20261024):
    rng = random.Random(seed)
    n = len(xs)
    if n == 0:
        return None
    boots = []
    for _ in range(B):
        sample = [xs[rng.randrange(n)] for _ in range(n)]
        boots.append(median(sample))
    boots.sort()
    lo = boots[int(0.025 * B)]
    hi = boots[min(int(0.975 * B), B - 1)]
    return {
        "median": round(median(xs), 6),
        "ci95_lo": round(lo, 6),
        "ci95_hi": round(hi, 6),
        "includes_0": lo <= 0 <= hi,
        "B": B,
        "seed": seed,
    }


lr_summary = {
    "note": "YoY continuous companion. NOT a locked adopt rule. e_naive=|y| (zero baseline). Negative log_err_ratio => model better.",
    "bootstrap_caveat": "n=5 (include) / n=4 (exclude) bootstrap is weak; resamples same points only.",
    "forms": {},
}
for f in FORMS:
    full = by_form[f]
    excl = [x for x in by_form[f] if x["t"] != 2025]
    blocks = {}
    for key, recs in (("include_2025", full), ("exclude_2025", excl)):
        xs = [x["log_err_ratio"] for x in recs]
        blocks[key] = {
            "n": len(xs),
            "years_neg_ratio": sum(1 for x in recs if x["model_better"]),
            "median_log_err_ratio": round(median(xs), 6),
            "mean_log_err_ratio": round(mean(xs), 6),
            "min": round(min(xs), 6),
            "max": round(max(xs), 6),
            "boot_median_ci95": boot_median_ci(xs),
        }
    # sign agreement
    mi = blocks["include_2025"]["median_log_err_ratio"]
    me = blocks["exclude_2025"]["median_log_err_ratio"]
    ci_i = blocks["include_2025"]["boot_median_ci95"]["includes_0"]
    ci_e = blocks["exclude_2025"]["boot_median_ci95"]["includes_0"]
    sign_i = 0 if mi == 0 else (1 if mi > 0 else -1)
    sign_e = 0 if me == 0 else (1 if me > 0 else -1)
    blocks["sign_agreement"] = {
        "median_sign_agree": sign_i == sign_e,
        "ci_includes_0_agree": ci_i == ci_e,
        "both_agree": (sign_i == sign_e) and (ci_i == ci_e),
    }
    lr_summary["forms"][f] = blocks

with (OUT / "yoy_log_err_ratio.csv").open("w", newline="") as fh:

def reading(b):
    """Language rule: 検出できず vs naiveより悪い."""
    if b["includes_0"]:
        return "検出できず／判断保留"
    if b["median"] > 0:
        return "モデルが悪い"
    if b["median"] < 0:
        return "モデルが良い"
    return "検出できず／判断保留"
